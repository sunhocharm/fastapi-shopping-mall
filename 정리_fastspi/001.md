## product 와 product_images 를 별도로 읽어서 합치는 방법

```Python
@router.get("/{product_id}/edit", response_model=ProductResponse)
async def edit_product_page(
    request: Request,
    product_id: int,
    category_service: CategoryServiceDep,
    product_service: ProductServiceDep,
    admin: CurrentAdminUser,
    templates: TemplatesDep
):
    logger.info("\n\n\t[[ edit_product_page ]]\n\n")

    try:
        product_only = await product_service.get_product_only_by_id(product_id)
        product_images_only = await product_service.get_product_images_by_product_id(product_id)
    except Exception:
        return RedirectResponse(url="/admin/products", status_code=303)

    # 2. ORM 객체를 딕셔너리로 변환 후 이미지 리스트 주입 (지연 로딩 에러 방지)
    product_dict = product_only.model_dump()
    product_dict["images"] = [image.model_dump() for image in product_images_only]

    # category_id가 있지만 category 객체가 쿼리되지 않은 경우를 위해 처리 (선택)
    if "category" not in product_dict:
        product_dict["category"] = None

    # 3. 딕셔너리 데이터를 기반으로 안전하게 검증 및 변환
    product = ProductResponse.model_validate(product_dict)

    categories = await category_service.get_all_categories()
    return templates.TemplateResponse(
        request=request,
        name="admin/products/form.html",
        context={"product": product, "categories": categories, "errors": {}}
    )
```
